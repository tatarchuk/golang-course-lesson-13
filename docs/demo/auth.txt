# Lesson 13, Part 1.8: token authentication, JWT (demo README 4.6; slides 32-33), 2026-10-27.
# Working directory: libratrack-grpc-demo/. The server is restarted with -auth. Tokens are printed masked
# (session helper 'mask': first 10 characters only); no JWT is stored in this repository (CI token guard).
# curl gets -sS (no progress meter outside a terminal), everything else as written in the README.

--- terminal 1: server with token authentication ---
$ go run ./cmd/inventory-server -auth        # keeps running; its output is appended below

--- terminal 2: mint the demo tokens (HS256, default demo secret, 1 hour) ---
$ export LIB=$(go run ./cmd/mint-token -sub anna -role librarian)
(exit 0)

$ export RDR=$(go run ./cmd/mint-token -sub oleh -role reader)
(exit 0)

$ mask LIB
LIB=eyJhbGciOi...[masked: 148 characters, 3 dot-separated parts]
(exit 0)

$ mask RDR
RDR=eyJhbGciOi...[masked: 144 characters, 3 dot-separated parts]
(exit 0)

--- terminal 2: the README 4.6 checks ---
$ go run ./cmd/inventory-client get copy-1
RPC помилка: code=Unauthenticated message="missing token"
exit status 1
(exit 1)

$ go run ./cmd/inventory-client -token "$RDR" get copy-1
{
  "id":  "copy-1",
  "book_id":  "book-1",
  "inventory_number":  "INV-0001",
  "status":  "AVAILABLE"
}
(exit 0)

$ go run ./cmd/inventory-client -token "$RDR" register b INV-1
RPC помилка: code=PermissionDenied message="role \"reader\" is not allowed to call /inventory.v1.InventoryService/RegisterCopy"
exit status 1
(exit 1)

$ go run ./cmd/inventory-client -token "$RDR" bulk book-3 2
RPC помилка: code=PermissionDenied message="role \"reader\" is not allowed to call /inventory.v1.InventoryService/BulkRegisterCopies"
exit status 1
(exit 1)

$ go run ./cmd/inventory-client -token "$LIB" register b INV-1
{
  "id":  "copy-4",
  "book_id":  "b",
  "inventory_number":  "INV-1",
  "status":  "AVAILABLE"
}
(exit 0)

--- terminal 2: an expired token (-ttl=-1m) is rejected too ---
$ export OLD=$(go run ./cmd/mint-token -sub oleh -role reader -ttl=-1m)
(exit 0)

$ go run ./cmd/inventory-client -token "$OLD" get copy-1
RPC помилка: code=Unauthenticated message="invalid token"
exit status 1
(exit 1)

--- terminal 2: grpcurl with the token in metadata ---
$ grpcurl -plaintext -H "authorization: Bearer $RDR" -d '{"id":"copy-1"}' localhost:50051 inventory.v1.InventoryService/GetCopy

  "id": "copy-1",
  "bookId": "book-1",
  "inventoryNumber": "INV-0001",
  "status": "AVAILABLE"

(exit 0)
